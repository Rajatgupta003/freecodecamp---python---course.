def number_pattern(n):
    # Check if the argument is an integer (handling booleans since they are a subclass of int in Python)
    if isinstance(n, bool):
        return "Argument must be an integer value."
    if not isinstance(n, int):
        return "Argument must be an integer value."
    
    # Check if the integer is less than 1
    if n < 1:
        return "Argument must be an integer greater than 0."
    
    # Use a for loop to build the space-separated string from 1 through n
    result = []
    for i in range(1, n + 1):
        result.append(str(i))
        
    return " ".join(result) # Or simply " ".join(...)
